function safeText(v,max){
  return typeof v==="string"||typeof v==="number"
    ?String(v).slice(0,max):"";
}

function normalizeLogo(raw,mimeHint){
  if(!raw)return "";

  let value=raw;
  let mime=mimeHint||"image/png";

  if(typeof raw==="object"){
    mime=raw.mimeType||raw.contentType||raw.type||mime;
    value=raw.dataUrl||raw.dataURL||raw.base64||
      raw.data||raw.url||raw.src||"";
  }

  if(typeof value!=="string")return "";
  value=value.trim();

  if(/^data:image\/(png|jpeg|jpg|webp|gif);base64,/i.test(value))
    return value;

  if(/^https:\/\/[^\s]+$/i.test(value))
    return value;

  const bare=value.replace(/\s/g,"").replace(/^base64,/i,"");

  if(bare&&/^[A-Za-z0-9+/]+={0,2}$/.test(bare)){
    if(!/^image\/(png|jpeg|jpg|webp|gif)$/i.test(mime))
      mime="image/png";
    return "data:"+mime+";base64,"+bare;
  }

  return "";
}

function parseDraft(raw){
  if(!raw||typeof raw!=="object")return null;

  const name=safeText(raw.name||raw.tokenName,100).trim();
  const symbol=safeText(raw.symbol||raw.tokenSymbol,10)
    .trim().toUpperCase();
  const supply=safeText(raw.supply||raw.totalSupply,40).trim();
  const decimals=Number(raw.decimals);
  const description=safeText(raw.description,1000).trim();

  const logoRaw=raw.logoData||raw.logoDataUrl||raw.logoBase64||
    raw.logoUrl||raw.logoURL||raw.logo||raw.image||raw.tokenLogo||"";

  const logoData=normalizeLogo(
    logoRaw,
    raw.logoMimeType||raw.mimeType
  );

  if(
    name.length<2||
    !/^[A-Z0-9._-]{2,10}$/.test(symbol)||
    !/^[1-9]\d*$/.test(supply)||
    supply.length>31||
    !Number.isInteger(decimals)||
    decimals<0||decimals>9||
    (raw.network&&raw.network!=="mainnet")
  )return null;

  try{
    const amount=BigInt(supply)*(10n**BigInt(decimals));
    if(amount<=0n||amount>10n**30n)return null;
  }catch(_){
    return null;
  }

  if(logoData.startsWith("data:")&&logoData.length>1400000)
    return null;

  if(logoRaw&&!logoData)return null;

  return {
    name,
    symbol,
    supply,
    decimals,
    description,
    logoData,
    network:"mainnet"
  };
}

function formatNanoTon(nano){
  if(!/^\d+$/.test(String(nano)))
    throw new Error("Backend returned an invalid fee amount.");

  const n=BigInt(String(nano));
  const whole=n/1000000000n;
  const frac=(n%1000000000n).toString()
    .padStart(9,"0").replace(/0+$/,"");

  return whole.toString()+(frac?"."+frac:"")+" TON";
}

async function createOrder(){
  if(!state.draft||!state.wallet||!state.address)
    throw new Error("Connect your wallet and complete token details first.");

  if(!connectedMainnet())
    throw new Error("Switch your wallet to TON Mainnet.");

  if(!$("VX03_accept").checked)
    throw new Error("Accept the fee policy to continue.");

  const token={
    name:state.draft.name,
    symbol:state.draft.symbol,
    supply:state.draft.supply,
    decimals:state.draft.decimals,
    description:state.draft.description
  };

  if(state.draft.logoData)token.logoData=state.draft.logoData;

  const data=await request("/api/orders",{
    method:"POST",
    headers:{"Content-Type":"application/json"},
    body:JSON.stringify({
      walletAddress:state.address,
      deploymentNetwork:"mainnet",
      token
    })
  });

  if(!data.order||!data.order.id)
    throw new Error("Backend did not return an order ID.");

  const order=data.order;
  const nano=order.amountNano;

  if(
    nano===undefined||nano===null||
    !/^[1-9]\d*$/.test(String(nano))
  ){
    throw new Error("Backend order has no valid fee amount. Payment blocked.");
  }

  if(order.feeTon!==undefined&&order.feeTon!==null){
    if(!/^\d+(?:\.\d{1,9})?$/.test(String(order.feeTon)))
      throw new Error("Backend returned an invalid TON fee.");

    const parts=String(order.feeTon).split(".");
    const expected=BigInt(parts[0])*1000000000n+
      BigInt((parts[1]||"").padEnd(9,"0")||"0");

    if(expected!==BigInt(String(nano)))
      throw new Error("Order fee fields do not match. Payment blocked.");
  }

  state.order={...order,walletAddress:state.address};

  state.paymentReady=!!(
    data.payment&&
    data.payment.paymentRequestEnabled===true&&
    data.payment.paymentVerificationEnabled===true
  );

  $("VX03_feeAmount").textContent=formatNanoTon(nano);

  show("VX03_payService",state.paymentReady);
  show("VX03_verifyService",state.paymentReady);

  status(
    "VX03_serviceStatus",
    state.paymentReady
      ?"Order created. Check the amount and recipient in your wallet before sending."
      :"Order created, but backend payment requests or verification are disabled."
  );

  refreshButtons();
}

async function getServicePayment(){
  if(!state.order)
    throw new Error("Create an order first.");

  if(!state.paymentReady)
    throw new Error("Backend payment requests and verification are not both enabled.");

  if(
    !connectedMainnet()||
    state.address!==state.order.walletAddress
  ){
    throw new Error("Reconnect the wallet that created this order.");
  }

  const p=await request(
    "/api/orders/"+encodeURIComponent(state.order.id)+"/payment-request"
  );

  if(p.network!==MAINNET)
    throw new Error("Payment request is not TON Mainnet.");

  if(String(p.amountNano)!==String(state.order.amountNano))
    throw new Error("Payment amount does not match this order. Payment blocked.");

  if(!p.address||typeof p.address!=="string")
    throw new Error("Backend payment recipient address is missing.");

  if(
    p.validUntil===undefined||p.validUntil===null||
    !Number.isFinite(Number(p.validUntil))||
    Number(p.validUntil)<=Math.floor(Date.now()/1000)
  ){
    throw new Error("Payment request is expired or invalid.");
  }

  if(
    p.payload!==undefined&&p.payload!==null&&
    typeof p.payload!=="string"
  ){
    throw new Error("Backend returned an invalid payment payload.");
  }

  return {
    ...p,
    validUntil:Number(p.validUntil),
    payload:p.payload||""
  };
}

async function verifyService(){
  if(!state.order)
    throw new Error("Create an order first.");

  if(
    !connectedMainnet()||
    state.address!==state.order.walletAddress
  ){
    throw new Error("Reconnect the wallet that created this order.");
  }

  const data=await request(
    "/api/orders/"+encodeURIComponent(state.order.id)+"/verify",
    {
      method:"POST",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify({walletAddress:state.address})
    }
  );

  if(data.status==="paid"&&data.paymentMarkedPaid===true){
    state.servicePaid=true;

    status("VX03_deployState","Payment verified");

    status(
      "VX03_serviceStatus",
      "Service payment verified by the backend."
    );

    status(
      "VX03_deploymentStatus",
      "Payment verified successfully by the backend."
    );

    advanceToDeployer();
    refreshButtons();
    return true;
  }

  status(
    "VX03_serviceStatus",
    data.message||"Payment is not confirmed yet. No success popup will be shown."
  );

  return false;
}

async function payService(){
  const p=await getServicePayment();

  await state.tonUI.sendTransaction({
    validUntil:p.validUntil,
    network:MAINNET,
    messages:[{
      address:p.address,
      amount:String(p.amountNano),
      payload:p.payload
    }]
  });

  status(
    "VX03_serviceStatus",
    "Transaction submitted. Waiting for backend verification…"
  );

  const ok=await verifyService();

  if(!ok){
    status(
      "VX03_serviceStatus",
      "Transaction submitted, but backend has not verified payment yet. Use Verify service-fee payment later."
    );
  }
}

  }
}

async function deployToken(){
 if(deploying||checking||completed||!backendEnabled||!orderData)return;
 deploying=true;refreshButton();
 try{
  setStatus("Rechecking deployment permission","Checking Render immediately before transaction preparation.","busy");
  const live=await api("/");
  backendEnabled=live.data.deploymentEnabled===true;
  if(!backendEnabled)throw new Error("Render has deployment disabled. No transaction was sent.");
  if(orderData.network!=="mainnet")throw new Error("Order is not for TON Mainnet.");
  const wallet=getConnectedWallet();
  if(wallet.address!==orderData.walletAddress)throw new Error("Connected wallet does not match the wallet that paid the service fee.");
  setStatus("Preparing transaction","Requesting a deployment transaction from the backend.","busy");
  const r=await api("/api/deployment/prepare",{
   method:"POST",headers:{"Content-Type":"application/json","Accept":"application/json"},
   body:JSON.stringify({orderId:orderData.orderId,walletAddress:wallet.address,network:"mainnet"})
  });
  const p=r.data,tx=p.transaction;
  if(p.deploymentEnabled!==true||p.status!=="prepared"||!tx||tx.network!==MAINNET||!Array.isArray(tx.messages)||!tx.messages.length)
   throw new Error(p.error||"The deployment service returned an invalid response.");
  setStatus("Wallet approval required","Review the transaction in your connected TON wallet.","busy");
  await wallet.ui.sendTransaction(tx);
  setStatus("Waiting for verification","Waiting for backend confirmation on-chain.","busy");
  const verified=await verifyDeployment(orderData.orderId,wallet.address);
  showSuccess(verified);completed=true;
 }catch(e){setStatus("Deployment not confirmed",e.message||"Deployment failed or is still pending.","error")}
 finally{deploying=false;refreshButton()}
}
