                <strong id="v2PreviewDecimals">9</strong>
              </div>
            </div>

          <p class="v2-preview-note">
            Preview only. This card is not a deployed token or a blockchain verification.
          </p>

          <div class="v2-fee">
            <div class="v2-fee-row">
              <span class="v2-fee-title">Platform Service Fee</span>
              <span class="v2-fee-amount">100 TON</span>

              The fee is displayed for review. Any payment must be verified
              by the backend before the workflow can proceed.

          <label class="v2-review-check" for="v2ConfirmDetails">
            <input id="v2ConfirmDetails" type="checkbox" required />
            <span>I have reviewed the token details and understand that the preview does not mean my token has been deployed.</span>
          </label>
        </aside>
      </div>

      <div class="v2-actions">
        <button class="v2-button" id="v2BackHome" type="button">
          <span aria-hidden="true">←</span> Back to Home
        </button>
        <button class="v2-button v2-button-primary" type="submit">
          Continue to Review <span aria-hidden="true">→</span>
        </button>
      </div>

      <div class="v2-message" id="v2FormMessage" role="status" aria-live="polite"></div>

    <p class="v2-bottom-note">
      MAGROXIN TOKEN LAB · Review your configuration before proceeding.
      Never share your wallet recovery phrase or private key.
    </p>

<script>
(function () {
  "use strict";

  var root = document.getElementById("VXTON_FORM_02");
  if (!root || root.dataset.v2Init === "1") return;
  root.dataset.v2Init = "1";

  var form = root.querySelector("#v2TokenForm");
  var nameInput = root.querySelector("#v2TokenName");
  var symbolInput = root.querySelector("#v2TokenSymbol");
  var descriptionInput = root.querySelector("#v2TokenDescription");
  var supplyInput = root.querySelector("#v2TokenSupply");
  var decimalsInput = root.querySelector("#v2TokenDecimals");
  var logoInput = root.querySelector("#v2TokenLogo");
  var confirmInput = root.querySelector("#v2ConfirmDetails");
  var message = root.querySelector("#v2FormMessage");

  if (
    !form || !nameInput || !symbolInput || !descriptionInput ||
    !supplyInput || !decimalsInput || !logoInput ||
    !confirmInput || !message
  ) {
    console.error("MAGROXIN: Required form elements are missing.");
    return;
  }

  var logoData = "";
  var logoMime = "";

  function validLogoUrl(value) {
    if (!value) return true;
    try {
      var url = new URL(value);
      return url.protocol === "https:" && !!url.hostname && !url.username && !url.password;
    } catch (e) {
      return false;
    }
  }

  function renderLogoPreview(value) {
    var preview = byId("v2LogoPreview");
    if (!preview) return;
    preview.replaceChildren();
    if (!value) {
      preview.textContent = (nameInput.value.trim() || "M").charAt(0).toUpperCase();
      return;
    }
    var img = document.createElement("img");
    img.alt = "Token logo preview";
    img.referrerPolicy = "no-referrer";
    img.onload = function () {
      if (preview.firstChild === img) {
        logoData = value;
        clearMessage();
      }
    };
    img.onerror = function () {
      if (preview.firstChild === img) {
        preview.textContent = (nameInput.value.trim() || "M").charAt(0).toUpperCase();
        logoData = "";
        showMessage("Logo URL could not load. Use a public HTTPS direct image link.", true);
      }
    };
    img.src = value;
    preview.appendChild(img);
  }

  function byId(id) {
    return root.querySelector("#" + id);
  }

  function showMessage(text, isError) {
    message.textContent = text;
    message.classList.add("v2-show");
    message.classList.toggle("v2-error", !!isError);
  }

  function clearMessage() {
    message.textContent = "";
    message.classList.remove("v2-show", "v2-error");
  }

  function formatSupply(value) {
    if (!value || !/^\d+$/.test(value)) return value || "—";

    try {
      return BigInt(value).toLocaleString("en-US");
    } catch (e) {
      return value;
    }
  }

  function updatePreview() {
    var name = nameInput.value.trim();
    var symbol = symbolInput.value.trim().toUpperCase();
    var description = descriptionInput.value.trim();
    var supply = supplyInput.value.trim();
    var decimals = decimalsInput.value;

    var previewName = byId("v2PreviewName");
    var previewSymbol = byId("v2PreviewSymbol");
    var previewDescription = byId("v2PreviewDescription");
    var previewSupply = byId("v2PreviewSupply");
    var previewDecimals = byId("v2PreviewDecimals");
    var descriptionCount = byId("v2DescriptionCount");
    var logoPreview = byId("v2LogoPreview");

    if (previewName) previewName.textContent = name || "Your Token";
    if (previewSymbol) previewSymbol.textContent = symbol ? "$" + symbol : "$SYMBOL";

    if (previewDescription) {
      previewDescription.textContent =
        description || "Your token description will appear here as you type.";
    }

    if (previewSupply) previewSupply.textContent = formatSupply(supply);
    if (previewDecimals) previewDecimals.textContent = decimals;

    if (descriptionCount) {
      descriptionCount.textContent = descriptionInput.value.length + " / 240";
    }

    if (logoPreview && !logoInput.value.trim()) {
      logoData = "";
      logoPreview.textContent = (name || "M").trim().charAt(0).toUpperCase();
    }
  }

  [
    nameInput,
    symbolInput,
    descriptionInput,
    supplyInput,
    decimalsInput
  ].forEach(function (el) {
    el.addEventListener("input", function () {
      clearMessage();

      if (el === symbolInput) {
        var start = symbolInput.selectionStart;
        var end = symbolInput.selectionEnd;

        var clean = symbolInput.value
          .replace(/[^a-zA-Z0-9]/g, "")
          .toUpperCase()
          .slice(0, 10);

        if (clean !== symbolInput.value) {
          symbolInput.value = clean;

          try {
            symbolInput.setSelectionRange(
              Math.min(start, clean.length),
              Math.min(end, clean.length)
            );
          } catch (e) {}
        }
      }

      updatePreview();
    });

    el.addEventListener("change", updatePreview);
  });

  logoInput.addEventListener("input", function () {
    clearMessage();
    var value = logoInput.value.trim();
    logoData = "";
    if (!value) {
      renderLogoPreview("");
      return;
    }
    if (!validLogoUrl(value)) {
      renderLogoPreview("");
      showMessage("Please enter a valid public HTTPS image URL.", true);
      return;
    }
    renderLogoPreview(value);
  });

  logoInput.addEventListener("change", function () {
    var value = logoInput.value.trim();
    if (value && validLogoUrl(value)) renderLogoPreview(value);
  });

  form.addEventListener("submit", function (event) {
    event.preventDefault();
    clearMessage();

    var name = nameInput.value.trim();
    var symbol = symbolInput.value.trim().toUpperCase();
    var description = descriptionInput.value.trim();
    var supply = supplyInput.value.trim();
    var decimals = Number(decimalsInput.value);

    if (name.length < 2 || name.length > 48) {
      showMessage(
        "Token name must be between 2 and 48 characters.",
        true
      );
      nameInput.focus();
      return;
    }

    if (!/^[A-Z0-9]{2,10}$/.test(symbol)) {
      showMessage(
        "Symbol must contain 2–10 letters or numbers.",
        true
      );
      symbolInput.focus();
      return;
    }

    if (description.length > 240) {
      showMessage(
        "Description must be 240 characters or fewer.",
        true
      );
      descriptionInput.focus();
      return;
    }

    if (!/^\d+$/.test(supply)) {
      showMessage("Enter a whole-number token supply.", true);
      supplyInput.focus();
      return;
    }

    var supplyBig;

    try {
      supplyBig = BigInt(supply);
    } catch (e) {
      showMessage("Invalid token supply.", true);
      supplyInput.focus();
      return;
    }

    if (supplyBig < 1n || supplyBig > 10n ** 30n) {
      showMessage("Supply must be between 1 and 10^30.", true);
      supplyInput.focus();
      return;
    }

    if ([0, 3, 6, 9].indexOf(decimals) === -1) {
      showMessage("Select a supported decimals value.", true);
      decimalsInput.focus();
      return;
    }

    var logoUrl = logoInput.value.trim();

    if (logoUrl && !validLogoUrl(logoUrl)) {
      showMessage("Logo must use a valid HTTPS URL.", true);
      logoInput.focus();
      return;
    }

    if (!confirmInput.checked) {
      showMessage(
        "Please confirm that you reviewed the token details.",
        true
      );
      confirmInput.focus();
      return;
    }

    var draft = {
      name: name,
      symbol: symbol,
      description: description,
      supply: supply,
      decimals: decimals,
      network: "mainnet",
      logoData: logoUrl || "",
      logoUrl: logoUrl || "",
      logoMime: "",
      platformFeeTon: "100"
    };

    /*
     * Keep the draft available if the final gadget initializes later.
     */
    window.VXTON_TOKEN_DRAFT = draft;
    window.MAGROXIN_TOKEN_DRAFT = draft;
    window.MGX_TOKEN_DRAFT = draft;

    /*
     * Notify any existing listeners without depending on these events
     * to preserve the draft.
     */
    window.dispatchEvent(
      new CustomEvent("vxton:token-draft", { detail: draft })
    );

    window.dispatchEvent(
      new CustomEvent("magroxin:token-draft", { detail: draft })
    );

    /*
     * Use the combined final deployer, Gadget 3 + Gadget 4.
     */
    var finalGadget = document.getElementById("MGX_FINAL_DEPLOYER");

    if (!finalGadget) {
      showMessage(
        "Token details are saved, but MGX_FINAL_DEPLOYER was not found. Add the combined final deployer and refresh the page.",
        true
      );
      return;
    }

    finalGadget.hidden = false;
    finalGadget.style.setProperty("display", "block", "important");
    finalGadget.setAttribute("aria-hidden", "false");

    /*
     * Let the final gadget read the saved draft, then scroll to it.
     */
    window.dispatchEvent(
      new CustomEvent("magroxin:open-review", { detail: draft })
    );

    window.dispatchEvent(
      new CustomEvent("vxton:open-review", { detail: draft })
    );

    showMessage(
      "Token details saved. Opening the MAGROXIN final deployer…",
      false
    );

    window.setTimeout(function () {
      finalGadget.scrollIntoView({
        behavior: "smooth",
        block: "start"
      });
    }, 180);
  });

  var backButton = byId("v2BackHome");

  if (backButton) {
    backButton.addEventListener("click", function () {
      window.dispatchEvent(new CustomEvent("vxton:back-to-home"));

      var home = document.getElementById("VXTON_HOME_01");

      if (home) {
        home.scrollIntoView({
          behavior: "smooth",
          block: "start"
        });
      }
    });
  }

  /*
   * Step 01 opens this form using this event.
   */
  window.addEventListener("vxton:open-token-form", function () {
    root.hidden = false;
    root.style.setProperty("display", "block", "important");
    root.setAttribute("aria-hidden", "false");
  });

  /*
   * Refresh the preview when the form initializes.
   */
  updatePreview();

})();
</script>


















</div>
