  await new Promise(resolve=>setTimeout(resolve,2500));
  const r=await api("/api/deployment/verify",{
   method:"POST",headers:{"Content-Type":"application/json","Accept":"application/json"},
   body:JSON.stringify({orderId,walletAddress,network:"mainnet"})
  });
  const d=r.data;
  if(d.deploymentEnabled===false){backendEnabled=false;refreshButton();throw new Error("Render disabled deployment. No deployment was confirmed.")}
  if(d.status==="deployed"&&d.verifiedOnChain===true&&d.tokenAddress)return d;
  if(d.status==="failed"||d.retryable===false)throw new Error(d.error||"Backend could not verify deployment.");
  setStatus("Verifying on TON Mainnet",d.message||"Waiting for on-chain confirmation…","busy");
 }
 throw new Error("Verification is still pending. No success popup was shown. Check status before retrying.");
}

async function deployToken(){

 deploying=true;refreshButton();

  setStatus("Rechecking deployment permission","Checking Render immediately before transaction preparation.","busy");

  const wallet=getConnectedWallet();
  if(wallet.address!==orderData.walletAddress)throw new Error("Connected wallet does not match the wallet that paid the service fee.");
  setStatus("Preparing transaction","Requesting a deployment transaction from the backend.","busy");
  const r=await api("/api/deployment/prepare",{
   method:"POST",headers:{"Content-Type":"application/json","Accept":"application/json"},
   body:JSON.stringify({orderId:orderData.orderId,walletAddress:wallet.address,network:"mainnet"})
  });
  const p=r.data,tx=p.transaction;
  if(p.deploymentEnabled!==true||p.status!=="prepared"||!tx||tx.network!==MAINNET||!Array.isArray(tx.messages)||!tx.messages.length)
   throw new Error(p.error||"The deployment service returned an invalid response.");
  setStatus("Wallet approval required","Review the transaction in your connected TON wallet.","busy");
  await wallet.ui.sendTransaction(tx);
  setStatus("Waiting for verification","Waiting for backend confirmation on-chain.","busy");
  const verified=await verifyDeployment(orderData.orderId,wallet.address);
  showSuccess(verified);completed=true;
 }catch(e){setStatus("Deployment not confirmed",e.message||"Deployment failed or is still pending.","error")}
 finally{deploying=false;refreshButton()}
}
